+def _find_func_body(source, pattern):
+    m = re.search(pattern, source)
+    if not m:
+        return ""
+    start = m.start()
+    # find opening brace
+    brace = source.find("{", m.end()-1)
+    if brace == -1:
+        return ""
+    depth = 0
+    for i in range(brace, len(source)):
+        if source[i] == "{":
+            depth += 1
+        elif source[i] == "}":
+            depth -= 1
+            if depth == 0:
+                return source[brace:i+1]
+    return ""
+
+def find_fn_body(source, name):
