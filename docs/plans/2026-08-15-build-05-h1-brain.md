```python
# tests/build05/reference_direction.py
from enum import Enum

class DIRECTION(Enum):
    STRONG_BEAR = -2
    BEAR = -1
    NEUTRAL = 0
    BULL = 1
    STRONG_BULL = 2

BULL_COMMIT = 0.45      # score to enter BULL
STRONG_BULL_COMMIT = 0.75
NEUTRAL_DROP = 0.20     # score to fall back to NEUTRAL
DWELL = 2               # consecutive bars at a candidate level before committing

def direction_enum(score, prev=DIRECTION.NEUTRAL, dwell=0):
    # clamped score
    s = max(-1.0, min(1.0, score))
    # candidate by thresholds
    if s >= STRONG_BULL_COMMIT: cand = DIRECTION.STRONG_BULL
    elif s >= BULL_COMMIT:      cand = DIRECTION.BULL
    elif s <= -STRONG_BULL_COMMIT: cand = DIRECTION.STRONG_BEAR
    elif s <= -BULL_COMMIT:      cand = DIRECTION.BEAR
    else:                        cand = DIRECTION.NEUTRAL
    # hysteresis: only commit to a *stronger* magnitude after dwell;
    # fall back to NEUTRAL when score drops below NEUTRAL_DROP
    mag = abs(cand.value)
    prev_mag = abs(prev.value)
    if mag == 0:
        return (cand, 0)
    if cand == prev:
        return (cand, min(dwell + 1, DWELL))
    if mag > prev_mag:
        if dwell + 1 >= DWELL:
            return (cand, 0)
        return (prev, dwell + 1)
    # weaker: allow immediate step down if score clearly left band
    if s < NEUTRAL_DROP and s > -NEUTRAL_DROP and cand == DIRECTION.NEUTRAL:
        return (DIRECTION.NEUTRAL, 0)
    return (cand, 0)
```

```python
# tests/build05/reference_momentum.py
from enum import Enum

class MOMENTUM(Enum):
    EXPANDING = 0
    STRONG = 1
    NORMAL = 2
    WEAK = 3
    DECAYING = 4

STRONG_THRESHOLD = 0.6
WEAK_THRESHOLD = 0.4
SLOPE_UP = 0.05
SLOPE_DOWN = -0.05
PERSISTENCE = 2

def momentum_enum(strength, slope, prev=MOMENTUM.NORMAL, persist=0):
    # strength in [0,1], slope in [-1,1]
    st = max(0.0, min(1.0, strength))
    sl = max(-1.0, min(1.0, slope))
    if st >= STRONG_THRESHOLD:
        cand = MOMENTUM.EXPANDING if sl >= SLOPE_UP else MOMENTUM.STRONG
    elif st >= WEAK_THRESHOLD:
        cand = MOMENTUM.NORMAL
    else:
        cand = MOMENTUM.DECAYING if sl <= SLOPE_DOWN else MOMENTUM.WEAK
    # persistence: require PERSISTENCE consecutive bars to switch out of EXPANDING/STRONG
    if prev in (MOMENTUM.EXPANDING, MOMENTUM.STRONG) and cand != prev:
        return prev if persist + 1 < PERSISTENCE else cand
    return cand
```

```python
# tests/build05/reference_volatility.py
from enum import Enum

class VOL_LEVEL(Enum):
    LOW = 0
    NORMAL = 1
    HIGH = 2
    EXTREME = 3

class VOL_QUALITY(Enum):
    HEALTHY = 0
    COMPRESSED = 1
    EXPANDING = 2
    CHAOTIC = 3
    SHOCK = 4

HIGH_RATIO = 1.5
EXTREME_RATIO = 2.0
LOW_RATIO = 0.7
LEVEL_DWELL = 2
QUALITY_GAP = 0.10
QUALITY_DWELL = 2

def volatility_level_enum(ratio, prev=VOL_LEVEL.NORMAL, dwell=0):
    if ratio >= EXTREME_RATIO: cand = VOL_LEVEL.EXTREME
    elif ratio >= HIGH_RATIO:  cand = VOL_LEVEL.HIGH
    elif ratio <= LOW_RATIO:   cand = VOL_LEVEL.LOW
    else:                      cand = VOL_LEVEL.NORMAL
    if cand == prev:
        return (cand, min(dwell + 1, LEVEL_DWELL))
    # only move to higher magnitude after dwell; allow step-down immediately
    if abs(cand.value) > abs(prev.value):
        if dwell + 1 >= LEVEL_DWELL:
            return (cand, 0)
        return (prev, dwell + 1)
    return (cand, 0)

def quality_enum(evidence, incumbent=(VOL_QUALITY.HEALTHY, 0.0, 0)):
    candidates = {
        VOL_QUALITY.HEALTHY: evidence.get("healthy", 0.0),
        VOL_QUALITY.COMPRESSED: evidence.get("compression", 0.0),
        VOL_QUALITY.EXPANDING: evidence.get("expansion", 0.0),
        VOL_QUALITY.CHAOTIC: evidence.get("chaos", 0.0),
        VOL_QUALITY.SHOCK: evidence.get("shock", 0.0),
    }
    best = max(candidates, key=candidates.get)
    inc_state, inc_conf, inc_dwell = incumbent
    if best != inc_state:
        if candidates[best] - inc_conf < QUALITY_GAP:
            return inc_state
        if inc_dwell + 1 < QUALITY_DWELL:
            return inc_state
    return best
```

```mql5
#ifndef ADAPTIVE_SURVIVAL_EA_MARKET_BRAIN_MQH
#define ADAPTIVE_SURVIVAL_EA_MARKET_BRAIN_MQH

#include "Types.mqh"

// Clamp helpers
double BrainClampSigned(const double v) { return MathMax(-1.0, MathMin(1.0, v)); }
double BrainClampUnit(const double v)  { return MathMax(0.0, MathMin(1.0, v)); }

// Net displacement over N bars / ATR (chronological arrays, index 0 = oldest)
double BrainDisplacement(const MqlRates &rates[], const double &atr[], const int count, const int bars)
{
   if(bars <= 0 || count < bars + 1) return 0.0;
   const double net = rates[count - 1].close - rates[count - 1 - bars].close;
   const double a = atr[count - 1];
   if(!(a > 0.0)) return 0.0;
   return net / a;
}

// Directional efficiency = |net| / sum(|bar-to-bar|)
double BrainEfficiency(const MqlRates &rates[], const int count, const int bars)
{
   if(bars <= 0 || count < bars + 1) return 0.0;
   double net = MathAbs(rates[count - 1].close - rates[count - 1 - bars].close);
   double path = 0.0;
   for(int i = count - bars; i < count; i++)
      path += MathAbs(rates[i].close - rates[i - 1].close);
   if(path <= 0.0) return 0.0;
   return net / path;
}

// Direction engine: consumes EMA20/EMA50 buffers (chronological), returns enum + score.
// Full implementation fills in the weighted evidence mean + hysteresis per BUILD 05 spec.
void DirectionEngine(const MqlRates &rates[], const double &emaFast[], const double &emaSlow[],
                     const double &atr[], const int count, DirectionResult &out)
{
   ZeroMemory(out);
   if(count < 3 || !BrainValidBuffers(atr, count))
   {
      out.state = DIRECTION_NEUTRAL;
      out.score = 0.0;
      out.valid = false;
      out.latestClosedH1 = rates[count - 1].time;
      return;
   }
   const double slopeFast = (emaFast[count-1] - emaFast[count-3]) / MathMax(atr[count-1], DBL_MIN);
   const double slopeSlow = (emaSlow[count-1] - emaSlow[count-3]) / MathMax(atr[count-1], DBL_MIN);
   const double displacement = BrainDisplacement(rates, atr, count, 20);
   const double efficiency = BrainEfficiency(rates, count, 20);
   const double positioning = (rates[count-1].close > emaFast[count-1] ? 1.0 : -1.0) * 0.5
                            + (rates[count-1].close > emaSlow[count-1] ? 1.0 : -1.0) * 0.5;
   // fixed weights (not inputs) per small-parameter-surface lock
   double raw = 0.30 * BrainTanh(slopeFast) + 0.25 * BrainTanh(slopeSlow)
              + 0.15 * BrainClampSigned(positioning)
              + 0.15 * BrainTanh(displacement) + 0.15 * BrainClampSigned(efficiency);
   out.score = BrainClampSigned(raw);
   out.state = DirectionHysteresis(out.score, DIRECTION_NEUTRAL, 0); // dwell tracked by caller state
   out.valid = true;
   out.latestClosedH1 = rates[count - 1].time;
}
```
